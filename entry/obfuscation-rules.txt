# ============================================================
# 生产配置：开启混淆 + 保护接口/持久化字段
# ============================================================

# === 启用混淆 ===
-enable-property-obfuscation
-enable-toplevel-obfuscation
-enable-export-obfuscation
-enable-filename-obfuscation
-compact
-remove-log

# === 保护持久化字段 ===
# PersistenceV2 序列化的 key 是属性名，混淆会改变 key
-keep-property-name
  token
  userName
  userId

# === 保护接口字段 ===
# 后端返回的 JSON 字段名，前端类型属性名必须保持一致
-keep-property-name
  code
  message
  data
  password
  isLogin
  isLoading
  inputContent
  pageUrl
  navigationId
  param
  result

# === 保护 HMRouter 关键字段 ===
# 虽然 HMRouter 已经 keep 了类名，但参数字段名也要保护
-keep-property-name
  interceptorName
  priority
  dialog
  singleton
